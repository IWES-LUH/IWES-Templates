%=== Thesis Type Configuration ==========================
% Command: \setthesistype{master|bachelor|ip|wai}
% Automatically sets all type-specific variables
\newcommand{\setthesistype}[1]{%
    \def\thesistype{#1}%
    % Master or Bachelor
    \ifboolexpr{ test {\ifdefstring{\thesistype}{master}} or test {\ifdefstring{\thesistype}{bachelor}} }{%
        \def\thesisdesc{Thesis submitted for the degree of}%
        \def\thesismodule{}%
        \newbool{showcoexaminer}\booltrue{showcoexaminer}%
        \newbool{showdegree}\booltrue{showdegree}%
    }{}%
    % Interdisciplinary Project
    \ifdefstring{\thesistype}{ip}{%
        \def\thesisdesc{Report for the module}%
        \def\thesismodule{Interdisciplinary Project}%
        \newbool{showcoexaminer}\boolfalse{showcoexaminer}%
        \newbool{showdegree}\boolfalse{showdegree}%
    }{}%
    % Scientific Work in Engineering
    \ifdefstring{\thesistype}{wai}{%
        \def\thesisdesc{Report for the module}%
        \def\thesismodule{Scientific Work in Engineering}%
        \newbool{showcoexaminer}\boolfalse{showcoexaminer}%
        \newbool{showdegree}\boolfalse{showdegree}%
    }{}%
}

% Warnings for missing DOI/URL/ISBN
\AtEveryBibitem{%
    % Articles: require DOI or URL
    \ifentrytype{article}{%
        \iffieldundef{doi}{%
            \iffieldundef{url}{%
                \PackageWarning{biblatex}{Missing DOI/URL for article: \thefield{entrykey}}%
            }{}%
        }{}%
    }{}%
    % Books: require ISBN or DOI
    \ifentrytype{book}{%
        \iffieldundef{isbn}{%
            \iffieldundef{doi}{%
                \PackageWarning{biblatex}{Missing ISBN/DOI for book: \thefield{entrykey}}%
            }{}%
        }{}%
    }{}%
}
